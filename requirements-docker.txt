# Dependencies for the Streamlit demo (Docker / examiner image).
# Includes xgboost + shap for the multisource model and waterfalls; no PyTorch / transformers.
#
# The numeric stack is pinned to the same versions as requirements.txt, because
# app.py trains at runtime: floating them lets the demo disagree with the numbers
# in results/*.json and PROJECT_REPORT.md 15.5.1. xgboost 3.4.1 requires Python
# >= 3.12, which is why the Dockerfile's base image is python:3.12-slim.

streamlit>=1.28.0
pandas==3.0.6
numpy==2.5.3
scipy==1.18.1
scikit-learn==1.9.1
# xgboost-cpu, not xgboost: identical version and import name, but without the
# nvidia-nccl-cu13 dependency (305 MB of CUDA) that the Linux xgboost wheel pulls
# in. This image is CPU-only, so that payload is dead weight in the layer.
xgboost-cpu==3.4.1
matplotlib>=3.8.0
shap==0.52.0
pyarrow>=14.0.0
